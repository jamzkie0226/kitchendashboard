function updateTimes() { var now = new Date(); // --- Local time (London) --- var localTime = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' }); var localDate = now.toLocaleDateString([], { weekday: 'short', year: 'numeric', month: 'short', day: 'numeric' }); document.getElementById('time-local').innerHTML = localTime; document.getElementById('date-local').innerHTML = localDate; // --- PH time (manual offset because iOS 9 ignores timeZone) --- var phNow = new Date(now.getTime() + (7 * 60 * 60 * 1000)); // +7 hours var phTime = phNow.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' }); var phDate = phNow.toLocaleDateString([], { weekday: 'short', year: 'numeric', month: 'short', day: 'numeric' }); document.getElementById('time-ph').innerHTML = phTime; document.getElementById('date-ph').innerHTML = phDate; } setInterval(updateTimes, 1000); updateTimes();
